@PostMapping("/orders")
@Transactional
public ResponseEntity<?> createOrder(@RequestBody OrderRequest request) {
    // 1. 检查库存
    Inventory inventory = inventoryRepository
        .findByDeviceIdAndProductId(request.getDeviceId(), request.getProductId())
        .orElseThrow(() -> new RuntimeException("库存不存在"));

    if (inventory.getQuantity() < request.getQuantity()) {
        return ResponseEntity.badRequest().body("库存不足");
    }

    // 2. 创建订单
    Order order = new Order();
    order.setOrderNo(generateOrderNo());
    order.setUserId(request.getUserId());
    order.setDeviceId(request.getDeviceId());
    order.setProductId(request.getProductId());
    order.setQuantity(request.getQuantity());
    order.setAmount(request.getAmount());
    order.setPaymentMethod(request.getPaymentMethod());
    order.setOrderStatus(OrderStatus.PENDING);
    orderRepository.save(order);

    // 3. 扣减库存（分布式锁）
    String lockKey = "inventory:" + inventory.getId();
    RLock lock = redissonClient.getLock(lockKey);
    try {
        lock.lock();
        inventory.setQuantity(inventory.getQuantity() - request.getQuantity());

        // 4. 自动变换库存状态
        if (inventory.getQuantity() == 0) {
            inventory.setStatus(InventoryStatus.OUT_OF_STOCK);
        } else if (inventory.getQuantity() < inventory.getMaxQuantity() * 0.3) {
            inventory.setStatus(InventoryStatus.LOW_STOCK);
        }

        inventoryRepository.save(inventory);
    } finally {
        lock.unlock();
    }

    // 5. 发送MQTT出货指令
    mqttClient.publish("device/" + request.getDeviceId() + "/dispense",
        JsonUtils.toJson(order));

    return ResponseEntity.ok(order);
}
```

@PostMapping("/upload")
public ResponseEntity<String> uploadImage(@RequestParam("file") MultipartFile file) {
    // 验证文件
    if (!Arrays.asList("image/jpeg", "image/png").contains(file.getContentType())) {
        throw new RuntimeException("不支持的文件类型");
    }
    if (file.getSize() > 5 * 1024 * 1024) {
        throw new RuntimeException("文件大小不能超过5MB");
    }

    // 保存到MinIO
    String filename = UUID.randomUUID() + "_" + file.getOriginalFilename();
    String url = minioService.upload(file, filename);

    return ResponseEntity.ok(url);
}
```
